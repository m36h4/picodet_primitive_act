python - <<'PY'
import json

f = "output_nikon_eval/bbox.json"

with open(f) as x:
    p = json.load(x)

print("Number predictions:", len(p))
print("First prediction:")
print(p[0])

print("\nScore range:")
scores = [x["score"] for x in p]
print("min:", min(scores))
print("max:", max(scores))

print("\nFirst 10 scores:")
print(sorted(scores, reverse=True)[:10])
